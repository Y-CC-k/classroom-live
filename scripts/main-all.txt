  private async runMuteScript(action: 'get' | 'set', value: string): Promise<string> {
    return new Promise(resolvePromise => {
      const file = this.ensureMuteScript();
      if (!file) {
        resolvePromise('');
        return;

      execFile(
        'powershell.exe',
        [
          '-NoProfile',
          '-NonInteractive',
          '-ExecutionPolicy',
          'Bypass',
          '-File',
          file,
          '-Action',
          action,
          '-Value',
          value,
        ],
        { windowsHide: true, timeout: 10_000 },
        (error, stdout) => {
          if (error) {
            this.log(`[broadcast] ??單憭望?嚗?{error.message.split('\n')[0] ?? ''}`);
          }
          resolvePromise(String(stdout ?? '').trim());
        }
      );
    });
  }

  private ensureMuteScript(): string | null {
    if (this.muteScriptPath && existsSync(this.muteScriptPath)) return this.muteScriptPath;
    try {
      // ?? import ?踹??惜靘陷嚗serData ?桅?銝摰???      const electron = require('electron') as { app: { getPath: (name: string) => string } };

      return this.muteScriptPath;
    } catch (error) {
      this.log(`[broadcast] ?⊥?皞???單嚗?{error instanceof Error ? error.message : String(error)}`);
      return null;
    }
  }

export function detectLocalIp(): string {
  const candidates: string[] = [];

  for (const infos of Object.values(os.networkInterfaces())) {
    for (const info of infos ?? []) {
      if (info.family !== 'IPv4' || info.internal) continue;
      candidates.push(info.address);
    }
  }

  return candidates.find(isPrivate) ?? candidates[0] ?? os.hostname();
}

function isPrivate(address: string): boolean {
  const parts = address.split('.').map(part => Number.parseInt(part, 10));
  const a = parts[0] ?? 0;
  const b = parts[1] ?? 0;
  if (a === 10) return true;
  if (a === 192 && b === 168) return true;
  if (a === 172 && b >= 16 && b <= 31) return true;
  return false;
}

 * 霈?箸璈?蝬脣鞈?嚗??菟?璈嚗? *

export function detectNetworkInfo(): NetworkInfo {
  const candidates: { address: string; mac: string }[] = [];

  for (const infos of Object.values(os.networkInterfaces())) {
    for (const info of infos ?? []) {
      if (info.family !== 'IPv4' || info.internal) continue;
      const mac = normalizeMac(info.mac ?? '');
      if (!mac) continue;
      candidates.push({ address: info.address, mac });
    }
  }

  const primary = candidates.find(item => isPrivate(item.address)) ?? candidates[0];

  const allMacs: string[] = [];
  for (const item of candidates) {
    if (!allMacs.includes(item.mac)) allMacs.push(item.mac);
  }

  // 銝餌雯?⊥?蝚砌?嚗?撣怎垢???桐蜓?萄停?臬?嚗擗雯?∪?臬???  const macs = primary ? [primary.mac, ...allMacs.filter(mac => mac !== primary.mac)].slice(0, 4) : allMacs.slice(0, 4);

  return {
    mac: primary?.mac ?? '',
    macs,
    ip: primary?.address ?? '127.0.0.1',
  };
}

function runReg(args: string[]): Promise<{ code: number; stdout: string }> {
  return new Promise(resolve => {
    execFile('reg.exe', args, { windowsHide: true, timeout: 5000 }, (error, stdout) => {
      const code =
        error && typeof (error as { code?: unknown }).code === 'number'
          ? (error as { code: number }).code
          : error
            ? 1
            : 0;
      resolve({ code, stdout: stdout ?? '' });
    });
  });
}

/**
 * ? `NotifyIconSettings` 摨???????瑁?瑼楝敺?璈Ⅳ?? *
 * `reg query ... /s /v ExecutablePath` ?撓?粹?見嚗? *   HKEY_CURRENT_USER\Control Panel\NotifyIconSettings\10930024916559073208
 *       ExecutablePath    REG_SZ    D:\...\electron.exe
 */
async function listIconKeys(): Promise<Array<{ key: string; exePath: string }>> {
  const { code, stdout } = await runReg(['query', NODE_ROOT, '/s', '/v', 'ExecutablePath']);
  if (code !== 0) return [];

  const result: Array<{ key: string; exePath: string }> = [];
  let currentKey: string | null = null;

  for (const rawLine of stdout.split(/\r?\n/)) {
    const line = rawLine.trim();
    if (!line) continue;

    if (/^HKEY_/i.test(line)) {
      currentKey = line;
      continue;
    }
    if (!currentKey) continue;

    // ?澆?蝔梯???賣?望?嚗??頂蝯梯?蝟餃蔣?踴?    const match = /^ExecutablePath\s+REG_SZ\s*(.*)$/i.exec(line);
    if (match) {
      result.push({ key: currentKey, exePath: (match[1] ?? '').trim() });
      currentKey = null;
    }
  }

  return result;
}

function normalizeExePath(value: string): string {
  return value.replace(/\\+/g, '\\').replace(/["']/g, '').trim().toLowerCase();
}

function exec(cmd: string, args: string[], timeoutMs = 15_000): Promise<{ code: number; out: string }> {
  return new Promise(resolve => {
    execFile(cmd, args, { windowsHide: true, timeout: timeoutMs }, (error, stdout, stderr) => {
      const code = error
        ? typeof (error as { code?: number | string }).code === 'number'
          ? (error as { code: number }).code
          : ((error as { status?: number }).status ?? 1)
        : 0;
      resolve({
        code,
        out: `${stdout ?? ''}${stderr ?? ''}`.trim(),
      });
    });
  });
}

/**

 * ??255.255.255.255:8787 ??127.0.0.1:8787嚗撩??嗅敺?閬? * { t: 'classroom-announce', wsUrl }???timeoutMs ?抒?????? */
import { createSocket } from 'node:dgram';

export interface DiscoveredServer {
  wsUrl: string;
}

export function discoverServers(timeoutMs = 2000): Promise<DiscoveredServer[]> {
  return new Promise(resolve => {
    const found = new Set<string>();
    const udp = createSocket({ type: 'udp4', reuseAddr: true });
    let settled = false;

    const finish = (): void => {
      if (settled) return;
      settled = true;
      try {
        udp.close();
      } catch {
        /* 撌脤??停蝞? */
      }
      resolve([...found].map(wsUrl => ({ wsUrl })));
    };

    udp.on('message', msg => {
      try {
        const parsed = JSON.parse(String(msg)) as { t?: string; wsUrl?: string };
        if (parsed?.t === 'classroom-announce' && typeof parsed.wsUrl === 'string') {
          found.add(parsed.wsUrl);
        }
      } catch {
        /* 憯??蕭??*/
      }
    });

    udp.on('error', () => finish());

    udp.bind(() => {
      udp.setBroadcast(true);
      const packet = Buffer.from(JSON.stringify({ t: 'classroom-discover' }));
      udp.send(packet, 8787, '255.255.255.255');
      udp.send(packet, 8787, '127.0.0.1'); // ?葦璈璈?頝?server ?葫閰行?憓?      setTimeout(finish, Math.max(500, timeoutMs));
    });
  });
}

const ILLEGAL_NAME_CHARS = /[<>:"/\\|?*\u0000-\u001f]/;

function sanitizeFileName(raw: string): string {
  let name = path.win32.basename(String(raw)).trim();
  name = name
    .replace(ILLEGAL_NAME_CHARS, '_')
    .replace(/\u007f/g, '_')
    .trim();
  name = name.replace(/[. ]+$/, '');
  const stem = name.replace(/\.[^.]+$/, '').toUpperCase();
  if (/^(CON|PRN|AUX|NUL|COM[1-9]|LPT[1-9])$/.test(stem)) name = `_${name}`;
  return name || 'file';
}

function dirCandidates(dir: string): string[] {
  const abs = path.resolve(dir);
  const root = path.win32.parse(abs).root;
  if (abs.toLowerCase() !== root.toLowerCase()) return [abs];
  return [root, path.join(root, 'SETUP')];
}

function canWriteDir(dir: string): boolean {
  try {
    fs.mkdirSync(dir, { recursive: true });
    const probe = path.join(dir, `.perm-test-${process.pid}`);
    fs.writeFileSync(probe, '');
    fs.unlinkSync(probe);
    return true;
  } catch {
    return false;
  }
}

function pickSaveDir(): string | null {
  const configured = getConfig().fileSaveDir;
  const configuredList = configured ? dirCandidates(configured) : [];

  if (resolvedSaveDir && configuredList.length > 0) {
    const wantKeys = configuredList.map(dir => dir.toLowerCase());
    if (!wantKeys.includes(path.resolve(resolvedSaveDir).toLowerCase())) {
      for (const dir of configuredList) {
        if (canWriteDir(dir)) {
          resolvedSaveDir = dir;
          return dir;
        }
      }
    }
  }
  if (resolvedSaveDir && canWriteDir(resolvedSaveDir)) return resolvedSaveDir;

  const candidates: string[] = [...configuredList];
  for (const dir of dirCandidates(defaultFileSaveDir())) candidates.push(dir);
  candidates.push(app.getPath('downloads'));
  candidates.push(path.join(app.getPath('desktop'), 'SETUP'));

  const configuredRoot = configured ? path.win32.parse(path.resolve(configured)).root.toLowerCase() : null;
  const seen = new Set<string>();
  for (const dir of candidates) {
    const key = path.resolve(dir).toLowerCase();
    if (seen.has(key)) continue;
    seen.add(key);
    if (!canWriteDir(dir)) {
      console.log(`[file] ?嗆?鞈?憭曆??舐嚗閰虫?銝?嚗?{dir}`);
      continue;
    }
    resolvedSaveDir = dir;
    const usedRoot = path.win32.parse(path.resolve(dir)).root.toLowerCase();
    if (configuredRoot && usedRoot !== configuredRoot) {
      console.log(`[file] ?函蔡閮剖???fileSaveDir嚗?{configured}嚗??臬神嚗祕?瑼?剁?${dir}`);
    }
    return dir;
  }
  return null;
}

function uniqueTarget(dir: string, name: string): string {
  const ext = path.extname(name);
  const stem = name.slice(0, name.length - ext.length);
  for (let index = 0; index < 1000; index++) {
    const candidate = index === 0 ? path.join(dir, name) : path.join(dir, `${stem}(${index})${ext}`);
    try {
      fs.accessSync(candidate);
    } catch {
      return candidate;
    }
  }
  return path.join(dir, `${stem}-${Date.now()}${ext}`);
}

function failTransfer(transfer: FileTransfer, reason: string): void {
  if (transfer.error) return;
  transfer.error = reason;
  try {
    fs.closeSync(transfer.fd);
  } catch {
    // 撌脤??停蝞?
  }
  try {
    fs.unlinkSync(transfer.target);
  } catch {
    // ?芯??停蝞?
  }
  if (transfer.idleTimer) clearTimeout(transfer.idleTimer);
}

function armIdleTimer(id: string, transfer: FileTransfer): void {
  if (transfer.idleTimer) clearTimeout(transfer.idleTimer);
  transfer.idleTimer = setTimeout(
    () => {
      const ongoing = fileTransfers.get(id);
      if (!ongoing) return;
      failTransfer(ongoing, '?喲葉?瘀??葦璈葉甇Ｘ??暹?嚗?);
      fileTransfers.delete(id);
    },
    3 * 60 * 1000
  );
}

 * 皜征?嗆?鞈?憭暸?撅斤?????殷??葦蝡胯?皜??嗆?憭整??改??? * 摰靽風嚗瑼??冗?亥身?箇?蝣?桅?嚗?憒?D:\嚗??寞? SETUP 摮??冗嚗? * SETUP 銋?摮??交?蝯?蝯?隤文蝤??孵惜鞈??? */
export async function cleanFileDir(): Promise<{ ok: boolean; deleted: number; message?: string }> {
  let dir = pickSaveDir();
  if (!dir) return { ok: false, deleted: 0, message: '?嗆?鞈?憭曆??舐' };
  const resolved = path.resolve(dir);
  if (path.win32.parse(resolved).root.toLowerCase() === resolved.toLowerCase()) {
    const setupDir = path.join(resolved, 'SETUP');
    if (fs.existsSync(setupDir)) {
      dir = setupDir;
    } else {
      return { ok: false, deleted: 0, message: '?嗆?鞈?憭曄蝤??寧???箏??刻絲閬歇頝喲?皜?' };
    }
  }
  let deleted = 0;
  try {
    for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
      const full = path.join(dir, entry.name);
      try {
        if (entry.isDirectory()) fs.rmSync(full, { recursive: true, force: true });
        else fs.unlinkSync(full);
        deleted++;
      } catch {
        /* ?桐??憭望?嚗◤?蝑??仿?嚗?銝剜?湔 */
      }
    }
    console.log('[file] ?嗆?鞈?憭曉歇皜?嚗? + dir + '嚗? + deleted + ' ??');
    return { ok: true, deleted };
  } catch (error) {
    return { ok: false, deleted, message: error instanceof Error ? error.message : String(error) };
  }
}

interface InjectPayload {
  kind: 'mouse-move' | 'mouse-button' | 'mouse-wheel' | 'key' | 'text';
  x?: number;
  y?: number;
  button?: string;
  down?: boolean;
  dy?: number;
  vk?: number;
  text?: string;
}

function inject(payload: InjectPayload): { ok: boolean; message?: string } {
  if (!binding) return { ok: false, message: '??璅∠??芾??? };
  try {
    const metrics = binding.getScreenMetrics();
    switch (payload.kind) {
      case 'mouse-move': {
        // ?葦蝡舀迤閬?摨扳?嚗?..1嚗撠蜓閬撟???摮貊?璈祕??蝝?        const x = metrics.x + (Number(payload.x) || 0) * metrics.primaryWidth;
        const y = metrics.y + (Number(payload.y) || 0) * metrics.primaryHeight;
        binding.moveAbs(Math.round(x), Math.round(y));
        return { ok: true };
      }
      case 'mouse-button': {
        const name = String(payload.button ?? 'left');
        const btn = (['left', 'right', 'middle'].includes(name) ? name : 'left') as MouseButtonName;
        const x = metrics.x + (Number(payload.x) || 0) * metrics.primaryWidth;
        const y = metrics.y + (Number(payload.y) || 0) * metrics.primaryHeight;
        binding.mouseButton(btn, payload.down !== false, Math.round(x), Math.round(y));
        return { ok: true };
      }
      case 'mouse-wheel': {
        // ?葦蝡舫?詻???蝡臬雿 WHEEL_DELTA嚗?20嚗?        binding.mouseWheel((Number(payload.dy) || 0) * 120, 0);
        return { ok: true };
      }
      case 'key': {
        binding.key(Number(payload.vk) || 0, payload.down !== false);
        return { ok: true };
      }
      case 'text': {
        binding.typeText(String(payload.text ?? ''));
        return { ok: true };
      }
      default:
        return { ok: false, message: '?芰?釣?仿??? };
    }
  } catch (error) {
    return { ok: false, message: error instanceof Error ? error.message : String(error) };
  }
}

import fs from 'node:fs';

export function registerSystemCommandsIpc(): void {
  let shutdownPending = false;

  ipcMain.handle(IPC.requestShutdown, (_event, delaySec: unknown) => {
    const delay =
      typeof delaySec === 'number' && Number.isFinite(delaySec) && delaySec >= 0
        ? Math.min(Math.round(delaySec), 600)
        : 30;

    const cancel = delaySec === -1;
    if (cancel) {
      execFile('shutdown.exe', ['/a'], { windowsHide: true, timeout: 10_000 }, error => {
        if (error) console.error('[shutdown] ????憭望?嚗?賣???????璈?');
        else console.log('[shutdown] 撌脣?瘨?摰???');
      });
      return true;
    }

    if (shutdownPending) return true;
    execFile('shutdown.exe', ['/s', '/t', String(delay)], { windowsHide: true, timeout: 10_000 }, error => {
      if (error) {
        console.error(`[shutdown] ????憭望?嚗?岫嚗?${error.message.split('\n')[0] ?? ''}`);
        shutdownPending = false;
      } else {
        console.log(`[shutdown] 撌脫?摰?${delay} 蝘???`);
      }
    });
    shutdownPending = true;
    return true;
  });

  ipcMain.handle(IPC.requestReboot, (_event, delaySec: unknown) => {
    const delay =
      typeof delaySec === 'number' && Number.isFinite(delaySec) && delaySec >= 0
        ? Math.min(Math.round(delaySec), 600)
        : 10;

    if (delaySec === -1) {
      execFile('shutdown.exe', ['/a'], { windowsHide: true, timeout: 10_000 }, error => {
        if (error) console.error('[reboot] ????/??憭望?嚗?賣????????殷?');
        else console.log('[reboot] 撌脣?瘨?摰???/??');
      });
      return true;
    }

    execFile('shutdown.exe', ['/r', '/f', '/t', String(delay)], { windowsHide: true, timeout: 10_000 }, error => {
      if (error) console.error(`[reboot] ????憭望?嚗?岫嚗?${error.message.split('\n')[0] ?? ''}`);
      else console.log(`[reboot] 撌脫?摰?${delay} 蝘???璈);
    });
    return true;
  });

  ipcMain.handle(IPC.execProgram, (_event, rawPath: unknown, rawArgs: unknown) => {
    const target = String(rawPath ?? '').trim();
    if (!target) return { ok: false, message: '瘝???閬銵?瑼?' };
    if (!fs.existsSync(target) || !fs.statSync(target).isFile()) {
      console.log(`[exec] 瑼?銝??剁?${target}`);
      return { ok: false, message: `瑼?銝??剁?${target}` };
    }
    const args = Array.isArray(rawArgs) ? rawArgs.map(item => String(item)) : [];
    const ext = path.extname(target).toLowerCase();
    try {
      if (ext === '.bat' || ext === '.cmd') {
        spawn('cmd.exe', ['/c', target, ...args], {
          cwd: path.dirname(target),
          detached: true,
          stdio: 'ignore',
          windowsHide: false,
        }).unref();
      } else {
        spawn(target, args, {
          cwd: path.dirname(target),
          detached: true,
          stdio: 'ignore',
          windowsHide: false,
        }).unref();
      }
      console.log(`[exec] 撌脣???${target}嚗?{args.length} ???賂?`);
      return { ok: true };
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      console.error(`[exec] ??憭望?嚗?{message}`);
      return { ok: false, message };
    }
  });

  ipcMain.handle(IPC.getScreenSource, async () => {
    const { desktopCapturer } = await import('electron');
    const sources = await desktopCapturer.getSources({ types: ['screen'] });
    if (sources.length === 0) {
      return { error: '?曆??啣?瑕??撟? };
    }
    return { id: sources[0]!.id };
  });
}

export function registerUpdateIpc(): void {
  ipcMain.handle(IPC.applyUpdate, async (): Promise<{ ok: boolean; reason?: string }> => {
    const logPath = path.join(app.getPath('userData'), 'update-log.txt');
    const stamp = () => `[${new Date().toISOString()}]`;
    const updateLog = (line: string): void => {
      try {
        fs.mkdirSync(path.dirname(logPath), { recursive: true });
        fs.appendFileSync(logPath, `${stamp()} (main) ${line}\r\n`, 'utf8');
      } catch {
        /* 撖思??脣撠梁?鈭?*/
      }
    };

    const recent = getLastStoredFile();
    if (!recent || Date.now() - recent.at > 30 * 60 * 1000 || !recent.path.toLowerCase().endsWith('.zip')) {
      updateLog('??嚗?0 ???扳????函? .zip ????);
      console.log('[update] 瘝??舐??蝝?嚗???30 ???扳?圈? .zip ????');
      return { ok: false, reason: '30 ???扳????函? .zip ???? };
    }
    try {
      if (fs.statSync(recent.path).size < 32 * 1024) {
        updateLog(`??嚗?蝝?擃???嚗?{recent.path}嚗);
        console.log('[update] ????蝛?撠?<32KB嚗??文??箔??舐');
        return { ok: false, reason: '????蝛?撠??文??箔??舐' };
      }
    } catch (error) {
      updateLog(`??嚗?銝????${error instanceof Error ? error.message : String(error)}嚗);
      console.error(`[update] 霈銝?????航撌脰◤皜?嚗?${error instanceof Error ? error.message : String(error)}`);
      clearLastStoredFile();
      return { ok: false, reason: '霈銝?????航撌脰◤皜?嚗? };
    }

    const appRoot = path.resolve(process.execPath, '..');
    const appDir = path.join(appRoot, 'resources', 'app');
    const stage = path.join(app.getPath('temp'), `classroom-update-${Date.now()}`);
    updateLog(`??閫??嚗ip=${recent.path} appRoot=${appRoot}`);

    const runTar = (): Promise<void> =>
      new Promise((resolve, reject) => {
        const tarExe = path.join(process.env.SystemRoot ?? 'C:\\Windows', 'System32', 'tar.exe');
        const child = spawn(tarExe, ['-xf', recent.path, '-C', stage], { windowsHide: true, stdio: 'ignore' });
        const timer = setTimeout(() => reject(new Error('tar 閫???暹?嚗?0 蝘?')), 60_000);
        child.on('exit', code => {
          clearTimeout(timer);
          if (code === 0) resolve();
          else reject(new Error(`tar 蝯?蝣?${code}`));
        });
        child.on('error', e => {
          clearTimeout(timer);
          reject(e);
        });
      });

    const runPsFallback = (): Promise<void> =>
      new Promise((resolve, reject) => {
        const ps = `Expand-Archive -LiteralPath '${recent.path.replace(/'/g, "''")}' -DestinationPath '${stage.replace(/'/g, "''")}' -Force`;
        const enc = Buffer.from(ps, 'utf16le').toString('base64');
        const child = spawn(
          'powershell.exe',
          ['-NoProfile', '-NonInteractive', '-ExecutionPolicy', 'Bypass', '-EncodedCommand', enc],
          {
            windowsHide: true,
            stdio: 'ignore',
          }
        );
        const timer = setTimeout(() => reject(new Error('Expand-Archive ?暹?嚗?0 蝘?')), 90_000);
        child.on('exit', code => {
          clearTimeout(timer);
          if (code === 0) resolve();
          else reject(new Error(`powershell 蝯?蝣?${code}`));
        });
        child.on('error', e => {
          clearTimeout(timer);
          reject(e);
        });
      });

    try {
      fs.mkdirSync(stage, { recursive: true });
      try {
        await runTar();
      } catch (tarError) {
        updateLog(
          `tar 閫??憭望?嚗??PowerShell ?嚗?{tarError instanceof Error ? tarError.message : String(tarError)}嚗
        );
        await runPsFallback();
      }
      const staged = fs.readdirSync(stage);
      if (staged.length === 0) throw new Error('閫??敺摮?蝛箇?嚗ip ?撮??嚗?);
      fs.cpSync(stage, appDir, {
        recursive: true,
        filter: src => path.basename(src).toLowerCase() !== 'config.json',
      });
    } catch (error) {
      const msg = error instanceof Error ? error.message : String(error);
      updateLog(`EXPAND FAILED: ${msg}`);
      console.error('[update] EXPAND FAILED嚗?, msg);
      try {
        fs.rmSync(stage, { recursive: true, force: true });
      } catch {
        /* 皜??停蝞? */
      }
      return { ok: false, reason: `閫??憭望?嚗?{msg}` };
    }
    updateLog('expanded update OK');
    try {
      fs.rmSync(stage, { recursive: true, force: true });
    } catch {
      /* 皜??停蝞? */
    }

    updateLog('relaunched app');
    console.log('[update] expanded update OK嚗撠????啁?');
    app.relaunch();
    app.exit(0);
    return { ok: true };
  });
}

export interface WindowManagerDeps {

function applyAutoStart(enabled: boolean): void {
  if (!app.isPackaged) {
    return;
  }
  try {
    app.setLoginItemSettings({
      openAtLogin: enabled,
      args: [],
    });
    console.log(`[autostart] ?餃?芸???嚗?{enabled ? '撌脣??? : '撌脣???}`);
  } catch (error) {
    console.error('[autostart] 閮剖??餃?芸???憭望?', error);
  }

export function createWindow(deps: WindowManagerDeps): BrowserWindow {
  const window = new BrowserWindow({
    width: 1280,
    height: 720,
    minWidth: 640,
    minHeight: 400,
    show: false,
    backgroundColor: '#05070b',
    title: '?恕?湔 ??摮貊?璈?,
    autoHideMenuBar: true,
    icon: path.join(__dirname, '..', '..', 'build', 'icon.png'),
    webPreferences: {
      preload: path.join(__dirname, '..', 'preload', 'index.cjs'),
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: false,
      backgroundThrottling: false,
    },
  });

  window.once('ready-to-show', () => window.show());

  deps.enforcer.configure(window, line => console.log(line));

  window.webContents.on('did-start-loading', () => deps.enforcer.reset());

  let closePromptOpen = false;
  window.on('close', event => {
    if (quitting) return;
    event.preventDefault();
    if (closePromptOpen) return;

    closePromptOpen = true;

    void dialog
      .showMessageBox(window, {
        type: 'question',
        buttons: ['蝮桀蝟餌絞??, '蝯?摮貊?璈?],
        defaultId: 0,
        cancelId: 0,
        noLink: true,
        title: '??摮貊?璈?,

      })
      .then(({ response }) => {
        if (response === 1) {
          quitting = true;
          app.quit();
        } else {
          window.hide();
        }
      })
      .catch(() => {
        window.hide();

        closePromptOpen = false;

function send(channel: string, payload?: unknown): void {
  if (!mainWindow || mainWindow.isDestroyed()) return;
  mainWindow.webContents.send(channel, payload);
}

function notifyRenderer(command: HostCommand): void {
  send(EVENTS.command, command);
}

function buildTray(deps: WindowManagerDeps): void {
  const icon = nativeImage.createFromDataURL(`data:image/png;base64,${TRAY_ICON_PNG_BASE64}`);
  tray = new Tray(icon);
  tray.setToolTip('?恕?湔 ??摮貊?璈?);

  const menu = Menu.buildFromTemplate([
    {
      label: '憿舐內閬?',
      click: deps.onShowWindow,
    },
    {
      label: '???刻撟?,
      click: deps.onToggleFullscreen,
    },
    {
      label: '????',
      click: deps.onReconnect,
    },
    { type: 'separator' },
    {
      label: '蝯?',
      click: deps.onQuitApp,
    },
  ]);

  tray.setContextMenu(menu);
  tray.on('double-click', deps.onShowWindow);
}

export function registerWindowIpc(deps: WindowManagerDeps): void {
  ipcMain.on(IPC.hideToTray, deps.onHideToTray);
  ipcMain.on('student:show-window', deps.onShowWindow);
  ipcMain.on(IPC.quitApp, deps.onQuitApp);
}

export function initializeApp(): void {
  const gotLock = app.requestSingleInstanceLock();
  if (!gotLock) {
    app.quit();
    return;
  }

  app.setAppUserModelId('com.classroom.student');

  app.on('second-instance', () => {
    mainWindow?.show();
    mainWindow?.focus();
  });

  const enforcer = new BroadcastEnforcer();
  const locker = new InputLocker(line => console.log(line));

  // ?葦蝡舫?摰?閫??萇皛?嚗???hook 撠?頛詨嚗?  // 銋?甇?handler 蝻箏仃嚗?摰?隞斗?鋡恍?暺?璉?renderer 蝡?invoke ?
  // ?o handler registered??∩犖撖死嚗??函嚗璈?摰?甇文??函??  ipcMain.handle(IPC.setInputLock, (_event, on: unknown) => {
    locker.setActive(on === true);
    return locker.isActive();
  });

  const deps: WindowManagerDeps = {
    enforcer,
    locker,
    onShowWindow: () => {
      if (mainWindow?.isMinimized()) mainWindow.restore();
      mainWindow?.show();
      mainWindow?.focus();
    },
    onHideToTray: () => mainWindow?.hide(),
    onQuitApp: () => {
      quitting = true;
      app.quit();
    },
    onToggleFullscreen: () => {
      mainWindow?.setFullScreen(!mainWindow?.isFullScreen());
    },
    onReconnect: () => {
      notifyRenderer('reconnect');
    },
  };

  app.whenReady().then(async () => {
    const config = getConfig();

    createWindow(deps);
    buildTray(deps);
    registerWindowIpc(deps);

    applyAutoStart(config.autoStart);

    void promoteTrayIcon().then(result => {
      console.log(`[tray] ${result.message}`);
    });

    keepAwakeId = powerSaveBlocker.start('prevent-display-sleep');

    runStartupBrowserClean();

    if (config.autoFullscreen) {
      mainWindow?.setFullScreen(true);
    }

    app.on('activate', () => {
      if (!mainWindow) createWindow(deps);
      mainWindow?.show();
    });
  });

  app.on('before-quit', () => {
    quitting = true;
    locker.dispose();
    if (keepAwakeId !== null) {
      powerSaveBlocker.stop(keepAwakeId);
      keepAwakeId = null;
    }
  });

  app.on('window-all-closed', () => {
    if (isDev) app.quit();
  });
}

export function getMainWindow(): BrowserWindow | null {
  return mainWindow;
}

export function setQuitting(value: boolean): void {
  quitting = value;
